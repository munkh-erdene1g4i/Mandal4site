Firebase өгөгдлийн сангийн зөвшөөрөл (Rules) хаалттай байна.

surguuliinsite төслийн Firebase Console → Firestore Database → Rules хэсэгт доорх дүрмийг хуулж Publish дарна уу. Админ бүрийг Authentication-д бүртгээд, admins collection-д тухайн имэйлээр (жижиг үсгээр) баримт үүсгэнэ: 

rules_version = '2';

// Мандал 4-р сургуулийн сайт — Firestore Rules (хувилбар 3.21)
service cloud.firestore {
  match /databases/{database}/documents {

    // Админ = Firebase-д нэвтэрсэн, имэйл нь admins жагсаалтад байгаа хүн
    function isAdmin() {
      return request.auth != null
        && request.auth.token.email != null
        && exists(/databases/$(database)/documents/admins/$(request.auth.token.email.lower()));
    }

    // Админ жагсаалт: хүн зөвхөн өөрийнхөө бичлэгийг харна, сайтаас өөрчлөх боломжгүй
    match /admins/{email} {
      allow read: if request.auth != null && request.auth.token.email.lower() == email;
      allow write: if false;
    }

    // Сайтын агуулга: хүн бүр уншина, зөвхөн админ бичнэ
    match /events/{docId}       { allow read: if true; allow write: if isAdmin(); }
    match /posts/{docId}        { allow read: if true; allow write: if isAdmin(); }
    match /gallery/{docId}      { allow read: if true; allow write: if isAdmin(); }
    match /good_lessons/{docId} { allow read: if true; allow write: if isAdmin(); }
    match /settings/{docId} {
      allow read: if true;
      // attendance_totals-ыг багш нар ирц хадгалахдаа шинэчилдэг тул нээлттэй
      allow write: if isAdmin() || docId == 'attendance_totals';
    }

    // Ирц: өмнөх шигээ (багшийн кодоор)
    match /attendance/{docId} { allow read, write: if true; }

    // Талархлын гацуур
    function validWish(d) {
      return d.keys().hasOnly(['to', 'from', 'message', 'style', 'color', 'slot', 'ts', 'createdAt'])
        && d.keys().hasAll(['to', 'message', 'style', 'color', 'slot', 'ts', 'createdAt'])
        && d.to is string && d.to.size() >= 1 && d.to.size() <= 40
        && (!('from' in d) || (d.from is string && d.from.size() <= 40))
        && d.message is string && d.message.size() >= 2 && d.message.size() <= 240
        && d.style in ['ball', 'star', 'bell', 'candy', 'gift', 'snow', 'heart', 'drop']
        && d.color is string && d.color.matches('^#[0-9a-fA-F]{6}$')
        && d.slot is int && d.slot >= 0 && d.slot < 400
        && d.ts is number
        && d.createdAt == request.time;
    }
    match /tree_wishes/{wishId} {
      allow read: if true;
      allow create: if validWish(request.resource.data);
      allow update: if false;
      allow delete: if isAdmin();
    }

    // Санал хүсэлт: хүн бүр илгээнэ, зөвхөн админ уншиж/тэмдэглэж/устгана
    function validFeedback(d) {
      return d.keys().hasOnly(['category', 'message', 'name', 'contact', 'ts', 'read', 'createdAt'])
        && d.keys().hasAll(['category', 'message', 'ts', 'read', 'createdAt'])
        && d.category in ['Санал', 'Хүсэлт', 'Талархал', 'Гомдол']
        && d.message is string && d.message.size() >= 3 && d.message.size() <= 1000
        && (!('name' in d) || (d.name is string && d.name.size() <= 40))
        && (!('contact' in d) || (d.contact is string && d.contact.size() <= 60))
        && d.ts is number
        && d.read == false
        && d.createdAt == request.time;
    }
    match /feedback/{fbId} {
      allow create: if validFeedback(request.resource.data);
      allow read, delete: if isAdmin();
      allow update: if isAdmin()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['read'])
        && request.resource.data.read is bool;
    }
  }
}
